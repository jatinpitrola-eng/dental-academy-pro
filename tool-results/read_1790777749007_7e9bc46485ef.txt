    13→// Generate a CUID-like ID (since we're not using Prisma's auto-generation).
    14→function generateId(): string {
    15→  const timestamp = Date.now().toString(36);
    16→  const random = Math.random().toString(36).slice(2, 10);
    17→  return `c${timestamp}${random}`;
    18→}

    20→// Add createdAt/updatedAt to data if not present. If the table doesn't have
    21→// updatedAt, the insert will fail and the calling code should handle it.
    22→function addTimestamps(data: Record<string, unknown>): Record<string, unknown> {
    23→  const now = new Date().toISOString();
    24→  if (!data.createdAt) data.createdAt = now;
    25→  if (!data.updatedAt) data.updatedAt = now;
    26→  return data;
    27→}

    29→// Execute an insert. If it fails due to a missing column (updatedAt/createdAt),
    30→// retry without that column. Returns the inserted row if RETURNING is supported.
    31→async function safeInsert(table: string, data: Record<string, unknown>): Promise<Record<string, unknown> | null> {
    32→  // Ensure data has an id — many tables require it.
    33→  if (!data.id) data.id = generateId();
    34→  addTimestamps(data);
    35→  const tryInsert = async (d: Record<string, unknown>): Promise<Record<string, unknown> | null> => {
    36→    const keys = Object.keys(d).map(k => `"${k}"`).join(", ");
    37→    const placeholders = Object.keys(d).map(() => "?").join(", ");
    38→    const args = Object.values(d);
    39→    try {
    40→      const res = await client.execute({ sql: `INSERT INTO "${table}" (${keys}) VALUES (${placeholders}) RETURNING *`, args });
    41→      return res.rows[0] ? toCamel(res.rows[0] as Record<string, unknown>) : null;
    42→    } catch (e) {
    43→      const err = e as Error;
    44→      if (err.message.includes("updatedAt")) {
    45→        const filtered = { ...d };
    46→        delete filtered.updatedAt;
    47→        return tryInsert(filtered);
    48→      }
    49→      if (err.message.includes("createdAt")) {
    50→        const filtered = { ...d };
    51→        delete filtered.createdAt;
    52→        return tryInsert(filtered);
    53→      }
    54→      throw e;
    55→    }
    56→  };
    57→  return tryInsert(data);
    58→}

    61→// Helper: convert snake_case DB rows to camelCase objects (like Prisma does).
    62→function toCamel(row: Record<string, unknown>): Record<string, unknown> {
    63→  const out: Record<string, unknown> = {};
    64→  for (const [k, v] of Object.entries(row)) {
    65→    const camelKey = k.replace(/_([a-z])/g, (_, c) => c.toUpperCase());
    66→    out[camelKey] = v;
    67→  }
    68→  return out;
    69→}

    71→// Helper: build WHERE clause from a filter object.
    72→function buildWhere(filter: Record<string, unknown>): { sql: string; args: unknown[] } {
    73→  const parts: string[] = [];
    74→  const args: unknown[] = [];
    75→  for (const [k, v] of Object.entries(filter)) {
    76→    if (v === null || v === undefined) continue;
    77→    // Handle compound keys (e.g., studentId_courseId: { studentId, courseId }).
    78→    if (k.includes("_") && typeof v === "object" && v !== null && !Array.isArray(v)) {
    79→      const compound = v as Record<string, unknown>;
    80→      const isCompoundKey = Object.keys(compound).every(
    81→        (ck) => typeof compound[ck] === "string" || typeof compound[ck] === "number",
    82→      );
    83→      if (isCompoundKey) {
    84→        for (const [ck, cv] of Object.entries(compound)) {
    85→          parts.push(`"${ck}" = ?`);
    86→          args.push(cv);
    87→        }
    88→        continue;
    89→      }
    90→    }
    91→    if (typeof v === "object" && v !== null && !Array.isArray(v)) {
    92→      // Handle Prisma-like operators: { gt: value }, { lt: value }, etc.
    93→      const ops = v as Record<string, unknown>;
    94→      for (const [op, opVal] of Object.entries(ops)) {
    95→        const sqlOp = op === "gt" ? ">" : op === "lt" ? "<" : op === "gte" ? ">=" : op === "lte" ? "<=" : "=";
    96→        parts.push(`"${k}" ${sqlOp} ?`);
    97→        args.push(opVal);
    98→      }
    99→    } else {
   100→      parts.push(`"${k}" = ?`);
   101→      args.push(v);
   102→    }
   103→  }
   104→  const sql = parts.length > 0 ? " WHERE " + parts.join(" AND ") : "";
   105→  return { sql, args };
   106→}
